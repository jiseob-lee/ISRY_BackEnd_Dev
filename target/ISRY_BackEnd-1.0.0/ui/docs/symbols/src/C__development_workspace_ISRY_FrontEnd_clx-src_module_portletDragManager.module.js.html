  8 /*
  9  * 본 모듈은 폼 레이아웃 내에서 사용자가 구획을 드래그 앤 드랍하여 포틀릿의 특성을 구현하기 위해서 만들어진 공통 모듈입니다.
 10  * 포틀릿을 구현한 뒤 개인화를 위해 쿠키에 저장하여 관리할 수 있습니다.
 11  * 
 12  * [사용법]
 13  * 로드되는 모든 화면에 대해서 이벤트 캐치를 하지 않습니다.
 14  * 따라서 각 화면의 BodyLoad시점에서 글로벌 함수 출판된 createDragManager를 사용하고 사용자속성을 정의합니다.
 15  * 개인화를 하기위해서는 글로벌 함수로 출판된 portletIndividual을 사용합니다. 
 16  * 개인화와 관련된 함수는 글로벌 함수인 portletIndividual의 하위에 속해있습니다.
 17  * 
 18  * 1. 포틀릿을 적용할 그룹(폼레이아웃일때만 가능)을 선택한 뒤, 선택한 그룹의 사용자 속성으로 portlet 을 정의하고 값은 Y로 지정합니다.
 19  * 2. 해당 포틀릿이 한 화면에서 하나의 폼 레이아웃에서만 사용하는게 아닐 경우가 있을 수 있기 때문에, dataType이라는 사용자 속성을 사용하여 드래그소스,
 20  * 	   드랍타겟에 대한 범위를 제한할 수 있습니다. (단, dataType속성에 대한 값은 unique해야 합니다.)
 21  * 3. 화면이 로딩되었을 때, onBodyLoad이벤트리스너에서 출판된 메서드 createDragManager를 호출합니다.
 22  * 4. 드래그앤드랍을 빈공간에 할 경우, 타겟컨트롤이 없는경우에는 드랍이 불가능합니다.
 23  *     따라서 해당 구획에는 빈 그룹을 배치해야 가능합니다.
 24  * 
 25  * [사용가능한 메서드]
 26  * 1. createDragManager : 드래그앤드롭을 시작합니다.
 27  * 2. removeDragManager : 드래그앤드롭 객체를 삭제합니다.
 28  * 3. setDragManagerStop : 전체 포틀릿 적용을 해제합니다.
 29  * 4. setResizePortlet : 포틀릿을 확대/축소합니다. (마우스가 눌린 시점부터 발생하기 때문에 mouseDown 이벤트리스너에서 호출하여 사용합니다.)
 30  * 5. setCookie : 쿠키를 생성합니다.
 31  * 6. getCookie : 쿠키를 반환합니다. 반환되는 쿠키는 포틀릿의 constraint 형식으로 반환됩니다.
 32  * 7. deketeCookie : 쿠키를 삭제합니다.
 33  */

164 /**
165  * 드래그 시작하는 메서드
166  * @param {cpr.core.AppInstance} app 앱인스턴스
167  * @param {Boolean} pbPermitDupl? 드래그앤드롭 시 아이템간 중복 시 드롭여부(true:중복된 아이템과 위치 스위칭, false : 드롭방지), default true
168  * @param {Boolean} pbIncRow? 드래그 시 마지막 행 기준 한 행 추가여부(true/false), default false
169  * @param {Boolean} pbDrpOrgSz? 드래그앤드롭 시 소스컨트롤 크기유지 여부(true/false) default true
170  */
171 function createDragManager(app, pbPermitDupl, pbIncRow, pbDrpOrgSz){
172 	
173 	/* 드래그앤드롭시 중복허용 */
174 	if(pbPermitDupl != null) {
175 		mbPermitDupl = pbPermitDupl;
176 	}
177 	
178 	/*. 드래그 시 한 행씩 추가 */
179 	if(pbIncRow != null) {
180 		mbIncreaseRow = pbIncRow;
181 	}
182 	
183 	/* 드래그앤드롭시 소스컨트롤 크기유지 */
184 	if(pbDrpOrgSz != null) {
185 		mbDropOrginSize = pbDrpOrgSz;
186 	}
187 	
188 	var voApp = app;
189 	var voContainer = voApp.getContainer();
190 	
191 	var vaRecursiveCtrls = voContainer.getAllRecursiveChildren();
192 	var vaDraggableContainers = vaRecursiveCtrls.filter(function(each){
193 		if (each.userAttr(ATTR_USE_PORTLET)== "Y") {
194 			return each;
195 		}
196 	});
197 
198 	vaDraggableContainers.forEach(function(/** cpr.controls.Container*/each){
199 			var vsDataType = each.userAttr(ATTR_DATA_TYPE);
200 			each.getChildren().forEach(function(eachs){
201 				if(eachs.userAttr("isAlready") == "") {
202 					eachs.userAttr("isAlready","Y");
203 					_createDragSource(voApp, eachs, vsDataType);
204 				}
205 		});
206 		_createDropTarget(each, vsDataType);
207 	});
208 
209 }

211 /**
212  * 특정 그룹의 드래그앤드롭 속성을 삭제합니다.
213  * @param {cpr.controls.Container} pcContainer
214  */
215 function removeDragManager(pcContainer) {
216 
217 	if(pcContainer) {
218 		var vaRecursiveCtrls = pcContainer.getChildren();
219 		vaRecursiveCtrls.forEach(function(each){
220 			each.removeUserAttr("isAlready");
221 		});
222 		
223 		var vsDataType = pcContainer.userAttr(ATTR_DATA_TYPE);
224 		maDragSource.forEach(function(each,idx){
225 			if(each.dataType == vsDataType) {
226 				maDragSource[idx].dispose();
227 			}
228 		});
229 		
230 		maDropTarget.forEach(function(each, idx){
231 			var vcCtrl = each.control;
232 			if(vcCtrl && vcCtrl.userAttr(ATTR_DATA_TYPE) == vsDataType) {
233 				maDropTarget[idx].dispose();
234 			}
235 		});
236 	}
237 
238 }

286 /**
287  * 드래그소스를 만들어주는 함수
288  * @param {cpr.core.AppInstance} app
289  * @param {cpr.controls.UIControl} pcDragCtrl
290  * @param {String} psDataType
291  */
292 function _createDragSource(app, pcDragCtrl, psDataType) {
293 	
294 	var voRootApp = app.getRootAppInstance();
295 	var vnIncreateIndex = true;
296 	
297 	var dragSource = new cpr.controls.DragSource(pcDragCtrl, {
298 		options: {
299 			dataType: psDataType,
300 			threadhold: 10, // 10px만큼 이동해야 드래그시작으로 인식
301 		},
302 		
303 		onBeforeDragStart: function(context){
304 			if(mbPosFix && !mbResize) {
305 				var voActualRect = context.source.control.getActualRect();
306 				var voStartLoc = context.dragStartLocation;
307 				// 앱헤더 영역만 드래그 가능하도록 수정 (2021.10.06 추가)
308 				if(voStartLoc.y >= (voActualRect.top + mnFixArrange)) {
309 //				if(voStartLoc.x <= (voActualRect.right - mnFixArrange) || voStartLoc.y >= (voActualRect.top + mnFixArrange)) {
310 					context.cancel();
311 				}
312 			}	
313 		},
314 		
315 		onDragStart: function(context) {
316 			
317 			if (mbDraggable) {
318 				if(context.source.control.type == "container" && context.source.control.getChildrenCount() == 0) {
319 					context.cancel();
320 				} else {
321 					var actualRect = pcDragCtrl.getActualRect();
322 					pcDragCtrl.style.css("opacity", " 0.5");
323 	
324 					if (mbResize == true) {
325 						// Span
326 						context.cursor = "se-resize";
327 						mcFeedback = _createDragSourceFeedback(actualRect, pcDragCtrl.getParent());
328 						voRootApp.floatControl(mcFeedback, actualRect);
329 					} else {
330 						// Drag
331 						context.cursor = "grabbing";
332 						mcFeedback = _createDragSourceFeedback(actualRect, pcDragCtrl.getParent());
333 						var newActualRect = actualRect.getTranslated(context.dragDelta);
334 						voRootApp.floatControl(mcFeedback, cpr.controls.layouts.XYLayout.createConstraintWithRect(newActualRect));
335 					}
336 				}
337 
338 			} else {
339 				context.cancel();
340 			}
341 		},
342 		onDragMove: function(context) {
343 			
344 			var actualRect = context.source.control.getActualRect();
345 
346 			if (mbResize == true) {
347 				// Span
348 				context.cursor = "se-resize";
349 				maDragDelta = context.dragDelta;
350 
351 				var voFloatConstraint = _getDragDirection(actualRect, pcDragCtrl);
352 				voRootApp.floatControl(mcFeedback, voFloatConstraint);
353 			} else {
354 				// Drag
355 				context.cursor = "grabbing";
356 				var newRect = actualRect.getTranslated(context.dragDelta);
357 				voRootApp.floatControl(mcFeedback, cpr.controls.layouts.XYLayout.createConstraintWithRect(newRect));
358 				
359 				if(mbIncreaseRow== true) {
360 					var vcParent = pcDragCtrl.getParent();
361 					var voLayout = vcParent.getLayout();
362 					var voRow = voLayout.getRows();
363 					var voColumn = voLayout.getColumns();
364 				
365 					if(context.target == null) {
366 						var vnChildCount = vcParent.getChildrenCount();
367 						var voLastChildRect = voLayout.getChildrenByLayoutOrder()[vnChildCount-1].getActualRect();
368 						var voFeedbackRect= mcFeedback.getActualRect();
369 						
370 						/* 아래로 드래그시 새로운 행 추가 */
371 						if(voFeedbackRect.bottom > voLastChildRect.bottom && context.dragDelta.height > 0) {
372 							if(vnIncreateIndex == true) {
373 								voRow.push("1fr");
374 								voLayout.setRows(voRow);
375 								vcParent.setLayout(voLayout);
376 								
377 								vnIncreateIndex = false;
378 							}
379 						}
380 						createDragManager(app, mbPermitDupl);
381 					}
382 				}
383 			}
384 		},
385 		onDragEnd: function(context) {
386 			
387 			if (mbResize == true && context.target) { // Span
388 				_setFormSpan(pcDragCtrl);
389 			}
390 			
391 			// 마지막 row에 아무런 아이템이 없을경우 삭제
392 			if(mbIncreaseRow) {
393 				_removeBlankRow(pcDragCtrl);
394 			}
395 
396 			context.cursor = "";
397 			mcFeedback.dispose();
398 			mcFeedback = null;
399 			pcDragCtrl.style.removeStyle("opacity");
400 			
401 			mbResize = false;
402 			vnIncreateIndex = true;
403 		}
404 	});
405 
406 	maDragSource.push(dragSource);
407 }

410 /**
411  * 드래그소스의 부모객체를받아와서 부모객체에 드랍타겟을 만들어주는 함수
412  * @param {cpr.controls.Container} pcCtrlDrop
413  * @param {String} psDataType
414  */
415 function _createDropTarget(pcCtrlDrop, psDataType) {
416 
417 	var vcContainer = pcCtrlDrop;
418 	var voParentLayout = vcContainer.getLayout();
419 	
420 	var vcDropFeedback = null;
421 	function getDropFeedback () {
422 		if(vcDropFeedback == null) {
423 			vcDropFeedback = new cpr.controls.Output();
424 			vcDropFeedback.style.css("box-shadow", maTargetStyle["box-shadow"]);
425 		}
426 		return vcDropFeedback;
427 	}
428 	
429 	function hideDropFeedback () {
430 		if (vcDropFeedback) {
431 			vcDropFeedback.dispose();
432 			vcDropFeedback = null;
433 		}
434 	}
435 	
436 	/* 드래그 시 컨트롤의 rowIndex 위치 */
437 	var vnDragRowIndex = null;
438 	
439 	/* 드래그 시 컨트롤의 colIndex 위치 */
440 	var vnDragColIndex = null;
441 	
442 	var dropTarget = new cpr.controls.DropTarget(vcContainer, {
443 		isImportant: function(source) {
444 			return source.dataType == psDataType;
445 		},
446 		onDragMove: function(context) {
447 			var reference = voParentLayout.getInsertionReference(context.pointerLocation);
448 
449 			var voSourceCtrlRect = vcContainer.getConstraint(context.source.control);
450 			reference.rowSpan = voSourceCtrlRect.rowSpan ?  voSourceCtrlRect.rowSpan : 1;
451 			reference.colSpan = voSourceCtrlRect.colSpan ?  voSourceCtrlRect.colSpan : 1;
452 			
453 			if(vnDragRowIndex == null) {
454 				vnDragRowIndex = reference.rowIndex - voSourceCtrlRect.rowIndex;
455 			}
456 			if(vnDragColIndex == null) {
457 				vnDragColIndex = reference.colIndex - voSourceCtrlRect.colIndex;
458 			}
459 			reference.rowIndex -= vnDragRowIndex;
460 			reference.colIndex -= vnDragColIndex;
461 			
462 			if (mbResize == false && reference) {
463 				vcContainer.addChild(getDropFeedback(), reference);
464 			} else {
465 				hideDropFeedback();
466 			}
467 		},
468 		onDragLeave: function(context) {
469 			hideDropFeedback();
470 		},
471 		onDrop: function(context) {
472 			
473 			if (mbResize == false) { 
474 				
475 				// 타겟 스타일 변경
476 				hideDropFeedback();
477 				
478 				// Drag 일 경우에만 변경되 위치로 드롭
479 				var voSourceCtrlRect = vcContainer.getConstraint(context.source.control);
480 				var insertionReference = voParentLayout.getInsertionReference(context.pointerLocation);
481 				insertionReference.rowSpan = voSourceCtrlRect.rowSpan ?  voSourceCtrlRect.rowSpan : 1;
482 				insertionReference.colSpan = voSourceCtrlRect.colSpan ?  voSourceCtrlRect.colSpan : 1;
483 			
484 				//2020-06-15 추가
485 				insertionReference.rowIndex -= vnDragRowIndex;
486 				insertionReference.colIndex -= vnDragColIndex;
487 			
488 				var vcTargetCtrl = voParentLayout.findControls(insertionReference).filter(function(each) {
489 					// 자기자신은 타겟에서 제외
490 					return each != context.source.control;
491 				})
492 				var voTargetCtrlRect = vcTargetCtrl.length > 0 ? vcContainer.getConstraint(vcTargetCtrl[0]) : insertionReference;
493 				
494 				voSourceCtrlRect.rowSpan = voSourceCtrlRect.rowSpan? voSourceCtrlRect.rowSpan : 1;
495 				voSourceCtrlRect.colSpan = voSourceCtrlRect.colSpan? voSourceCtrlRect.colSpan : 1;
496 				voTargetCtrlRect.rowSpan = voTargetCtrlRect.rowSpan? voTargetCtrlRect.rowSpan : 1;
497 				voTargetCtrlRect.colSpan = voTargetCtrlRect.colSpan? voTargetCtrlRect.colSpan : 1;
498 				
499 				if(mbPermitDupl == true) {
500 					
501 					/* 중복허용 - 위치 스위칭 */

584 			}
585 			
586 			vnDragRowIndex = null;

588 		}

590 	
591 	maDropTarget.push(dropTarget);

678 /**
679  * 드래그 플로팅 constraint
680  * @param {cpr.geometry.Rectangle} poSourceRect
681  * @param {cpr.controls.UIControl} pcDragCtrl
682  * @return {cpr.controls.layouts.Constraint} constraint
683  */
684 function _getDragDirection (poSourceRect, pcDragCtrl) {
685 	
686 	var vcParent = pcDragCtrl.getParent();
687 	var voConstraint = vcParent.getConstraint(pcDragCtrl);
688 	
689 	var vsTop = poSourceRect.top + "px";
690 	var vsLeft = poSourceRect.left + "px";
691 	var vsWidth = (poSourceRect.width + maDragDelta.width) + "px";
692 	var vsHeight = (poSourceRect.height + maDragDelta.height) + "px";
693 	
694 	if(voConstraint.rowSpan > 1 || voConstraint.colSpan > 1) {
695 		// 확장된 영역에서 피드백 크기 감소
696 		vsTop = poSourceRect.top + "px";			
697 		vsLeft = poSourceRect.left + "px";			
698 		vsWidth = (poSourceRect.width + maDragDelta.width) + "px";
699 		vsHeight = (poSourceRect.height + maDragDelta.height) + "px";
700 	}
701 
702 	/* 포틀릿 위치 벗어나지 않도록 영역 제한 */
703 	if(vsTop.split("px")[0] < vcParent.getActualRect().top) { // 위
704 		vsTop = vcParent.getActualRect().top + "px";
705 		vsHeight = mcFeedback.getActualRect().height + "px";
706 	}
707 	
708 	if(vsLeft.split("px")[0] < vcParent.getActualRect().left) { // 왼쪽
709 		vsLeft = vcParent.getActualRect().left + "px";
710 		vsWidth = mcFeedback.getActualRect().width + "px";
711 	} 
712 	
713 	if((parseInt(vsWidth.split("px")[0])+parseInt(vsLeft.split("px")[0])) > vcParent.getActualRect().right) { // 오른쪽
714 		vsWidth = (vcParent.getActualRect().right - vsLeft.split("px")[0]) + "px";
715 	}
716 	
717 	if((parseInt(vsHeight.split("px")[0])+parseInt(vsTop.split("px")[0])) > vcParent.getActualRect().bottom) { // 아래
718 		vsHeight = (vcParent.getActualRect().bottom - vsTop.split("px")[0]) + "px";
719 	}
720 
721 	return {
722 		top : vsTop, 
723 		left : vsLeft,
724 		width : vsWidth,
725 		height : vsHeight
726 	};

730 /**
731  * 폼레이아웃 확대/축소(span)
732  * 타겟 영역의 절반 이상일 때만 확대/축소합니다.
733  * 폼레이아웃의 row와 column은 모두 같은 크기로만 구성되어있어야 합니다.
734  * @param {cpr.controls.Container} pcDragCtrl
735  */
736 function _setFormSpan (pcDragCtrl) {
737 	
738 	var vcParent = pcDragCtrl.getParent();
739 	var voConstraint = vcParent.getConstraint(pcDragCtrl);
740 	var voLayout = vcParent.getLayout();
741 
742 	var vnRowLen = voLayout.getRows().length;
743 	var vnColLen = voLayout.getColumns().length;
744 	var vnOneWidth = voLayout.getColumns()[0].indexOf("fr") > 0 ? vcParent.getActualRect().width / vnColLen : voLayout.getColumns()[0].split("px")[0];
745 	var vnOneHeight = voLayout.getRows()[0].indexOf("fr") > 0 ? vcParent.getActualRect().height / vnRowLen : voLayout.getRows()[0].split("px")[0];
746 	
747 	/* span 영역 */
748 	var voActualRect = pcDragCtrl.getActualRect();
749 	var voFeedbackRect = mcFeedback.getActualRect();
750 	
751 	// 1. rowSpan
752 	var vnRowSpan = parseInt(Math.round(voFeedbackRect.height / vnOneHeight));
753 	if(vnRowSpan == 0) vnRowSpan =1 ; 
754 	
755 	// 2. colSpan
756 	var vnColSpan = parseInt(Math.round(voFeedbackRect.width / vnOneWidth));
757 	if(vnColSpan == 0) vnColSpan =1 ; 
758 
759 	// 3. rowIndex
760 	var vnRowIndex = voConstraint.rowIndex;
761 	if(voConstraint.rowSpan != vnRowSpan) {
762 		var vnRowDiff = voConstraint.rowSpan - vnRowSpan;
763 		if(maDragDelta.height < 0 && vnRowSpan > voConstraint.rowSpan) {
764 			vnRowIndex = vnRowIndex + vnRowDiff;
765 		}
766 	}
767 
768 	// 4. colIndex
769 	var vnColIndex = voConstraint.colIndex;
770 	if(voConstraint.colSpan != vnColSpan) {
771 		var vnColDiff = voConstraint.colSpan - vnColSpan;
772 		if(maDragDelta.width < 0 && vnColSpan > voConstraint.colSpan) {
773 			vnColIndex = vnColIndex + vnColDiff;
774 		}
775 	}
776 	
777 	var vcDuplCtrl = voLayout.findControls({
778 		rowIndex: vnRowIndex,
779 		colIndex: vnColIndex,
780 		rowSpan: vnRowSpan,
781 		colSpan: vnColSpan
782 	}).filter(function(each){
783 		if(each.type != "container" && each != pcDragCtrl) {
784 			return each;
785 		}
786 	});
787 	
788 	if(vcDuplCtrl.length > 0) {
789 		alert("컨트롤 간 겹칠 수 없습니다.");
790 		return;
791 	}
792 	
793 	// 차트가 새로 그려질 수 있도록 삭제 후 추가
794 	vcParent.removeChild(pcDragCtrl);
795 	vcParent.addChild(pcDragCtrl, {
796 		rowIndex : vnRowIndex,
797 		colIndex : vnColIndex,
798 		rowSpan : vnRowSpan,
799 		colSpan :  vnColSpan
800 	});
801 }
