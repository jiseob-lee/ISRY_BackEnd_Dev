  8 /**
  9  * 두 문자열 배열을 비교합니다.
 10  * @param {String[]} a
 11  * @param {String[]} b
 12  */
 13 function equals(a, b) {
 14 	if (a === null && b === null) {
 15 		return true;
 16 	}
 17 	if (a === null || b === null) {
 18 		return false;
 19 	}
 20 
 21 	if (a.length !== b.length) {
 22 		return false;
 23 	}
 24 
 25 	return a.every(function(each, idx) {
 26 		return each == b[idx];
 27 	});
 28 }
 29 
 30 /**
 31  * @constructor
 32  * @param {cpr.controls.Container} container
 33  * @param {()=>void} resizeCallback
 34  */
 35 function Tracker(container, resizeCallback) {
 36 	this._container = container;
 37 	this._resizeCallback = resizeCallback;
 38 
 39 	/** @type cpr.controls.layouts.FormLayout */
 40 	this._layout = container.getLayout();
 41 
 42 	this._knownColumns = this._layout.getColumns().slice();
 43 	this._knownRows = this._layout.getRows().slice();
 44 }
 45 
 46 Tracker.prototype.start = function() {
 47 	this.onBeforeDraw = this.onBeforeDraw.bind(this);
 48 
 49 	// 컨테이너가 새로 그려질 때 마다 체크.
 50 	this._container.addEventListener("before-draw", this.onBeforeDraw);
 51 };
 52 
 53 /**
 54  * 마지막 확인 시점으로 부터 폼 레이아웃의 구획 크기가 변경되었는지 여부를 반환합니다.
 55  */
 56 Tracker.prototype.isResized = function() {
 57 	// 가로 영역이 변경 체크.
 58 	if (!equals(this._knownColumns, this._layout.getColumns())) {
 59 		return true;
 60 	}
 61 
 62 	// 세로 영역 변경 체크.
 63 	if (!equals(this._knownRows, this._layout.getRows())) {
 64 		return true;
 65 	}
 66 
 67 	return false;
 68 }
 69 
 70 /**
 71  * 컨테이너가 새로 그려질 때 마다 호출되는 핸들러.
 72  */
 73 Tracker.prototype.onBeforeDraw = function() {
 74 	if (this.isResized()) {
 75 		// 현재 크기 기억.
 76 		this._knownColumns = this._layout.getColumns().slice();
 77 		this._knownRows = this._layout.getRows().slice();
 78 
 79 		// 사용자 콜백 호출.
 80 		this._resizeCallback();
 81 	}
 82 };
 83 
 84 /**
 85  * 사용자가 폼 레이아웃 구분선을 드래그하여 리사이즈 한 경우의 콜백을 추가합니다.
 86  *  
 87  * @param {cpr.controls.Container} container 폼 레이아웃으로 구성된 그룹.
 88  * @param {()=>void)} callback 리사이즈 콜백.
 89  */
 90 globals.hookFormResize = function(container, callback) {
 91 	new Tracker(container, callback).start();
 92 }
